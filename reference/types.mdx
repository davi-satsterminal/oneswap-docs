All types are exported from `oneswap-v2-sdk`.

## Core

```ts
interface Account { id: string; username: string; walletId: string; createdAt?: string }

interface Token { symbol: string; admin: string; id: string; registryUrl?: string }

interface Balance { instrumentId: string; admin: string; amount: number; holdings: number; locked: number }

interface Offer {
  contractId: string; sender: string; receiver: string;
  amount: number; instrumentId: { admin: string; id: string }; status: string;
}

interface Pool {
  id: string; operator: string; poolParty: string;
  assetX: Token; assetY: Token;
  reserveX: number; reserveY: number;
  feeBps: number; totalLpTokens: number;
  name?: string | null; visible?: boolean;
  settlementMode?: 'atomic';   // v2 is atomic DvP only
  primary?: boolean;
}

interface Quote {
  amountOut: number; priceImpactPct?: number; swapFeeBps?: number;
  networkFeeIn?: number; effInput?: number;
  newReserveX: number; newReserveY: number;
  lpFee?: number; platformFee?: number; discountBps?: number;
}

interface SwapResult {
  updateId: string | null; amountOut: number;
  newReserveX: number; newReserveY: number;
  atomic?: boolean; intentId?: string; note?: string;
}

interface Position {
  poolId: string; lpTokens: number; sharePct: number;
  valueX: number; valueY: number; assetX?: string; assetY?: string;
}
```

## Interactive (self-custody)

```ts
/** An unsigned atomic-DvP transaction. Sign `preparedTransactionHash`. */
interface Prepared {
  preparedTransaction: string;
  preparedTransactionHash: string;
  costBytes?: number;
  quote?: Quote;
}

/** Pluggable signer — implement over any wallet / validator. */
interface WalletSigner {
  party: string;
  publicKeyFingerprint: string;
  signHash(preparedTransactionHash: string): string | Promise<string>;
}

/** One party's signature over a prepared transaction hash. */
interface Signature { party: string; signature: string; publicKeyFingerprint: string }

/** A transfer leg (the trader's side; pool/fee legs added by the operator). */
interface Leg { sender: string; receiver: string; instrumentId: { admin: string; id: string }; amount: string | number }

interface MandateParties { partyA: string; partyB: string }
```

## Constructor options

```ts
interface OneSwapOptions {
  session?: string;      // restore a custodial session token
  adminToken?: string;   // x-admin-token, for self-hosted admin/interactive endpoints
  slippageBps?: number;  // default custodial slippage (default 200 = 2%)
}
```
